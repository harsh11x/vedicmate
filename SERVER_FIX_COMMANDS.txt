# COPY AND PASTE THESE COMMANDS ON YOUR SERVER
# Run these commands on ubuntu@15.207.36.26

# 1. Navigate to project
cd ~/vedicmate

# 2. Pull latest changes
git pull

# 3. Go to admin directory
cd admin

# 4. Stop the admin panel
pm2 stop vedicmate-admin

# 5. CRITICAL: Delete the corrupted build cache
rm -rf .next

# 6. Verify .env.local exists and has correct values
cat .env.local
# Should show:
# NEXT_PUBLIC_API_URL=http://15.207.36.26:3001/api
# NEXT_PUBLIC_API_BACKEND=http://15.207.36.26:3001

# 7. Install dependencies (in case any are missing)
npm install --production

# 8. Build fresh
npm run build

# 9. Restart admin panel
pm2 restart vedicmate-admin

# 10. Save PM2 configuration
pm2 save

# 11. Check logs to verify it's working
pm2 logs vedicmate-admin --lines 20

# 12. Test the health endpoint
curl http://localhost:3001/api/health

# Done! Now refresh your browser at https://15.207.36.26:3000
